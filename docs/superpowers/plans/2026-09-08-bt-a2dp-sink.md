> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The EVKB is a Bluetooth speaker: an iPhone pairs with it, streams music over A2DP, and it comes out of the WM8962 headphone jack cleanly for as long as the phone plays, with the phone's volume buttons controlling the level.

**Architecture:** Mirror the source. New library units `bt/SbcDecoder` (the encoder's inverse), `bt/A2dpSink` (one inbound attempt as AVDTP acceptor + media receive) and `bt/BtSinkSession` (the discoverable/connectable idle policy) beside their source twins; the existing `Avdtp` acceptor gains a SNK personality and DelayReport sending, `Sdp` an AudioSink record, `Avrcp` absolute volume, `BtLink` an identity (class of device, name), inquiry scan and accept-unknown knobs. The example `audio/bt_sink_test` owns `AudioInputBluetooth`, which decodes into a PCM ring, feeds the audio graph, and servos the RT1176 audio PLL on the ring's fill so the SAI consumes at the phone's rate. One new QEMU gate (139): the fake peer becomes a SOURCE streaming the encoder tests' `sine.sbc`.

**Tech Stack:** C++11 (host tests: `c++ -std=c++11 -Wall -Wextra -Werror`; firmware: ARM GCC 10 + CMake), Python 3 (`hci_peer.py`), the tree's QEMU gates, LinkServer for silicon. Spec: `docs/superpowers/specs/2026-09-08-bt-a2dp-sink-design.md`. Linear: NEW-41.

```cpp
// SbcDecoder -- clean-room SBC DECODER from A2DP v1.3 section 12, the mirror of Sbc (the encoder): frame header
// and CRC-8 (Sbc::crc8), scale factors, the SAME loudness bit allocation (Sbc::allocateBits), dequantisation,
// joint-stereo un-mixing, and the 8-subband SYNTHESIS filterbank (12.6.4) inverting Sbc::analyse.  8 subbands,
// 4..16 blocks, all channel modes, bitpool 2..53, loudness allocation (the encoder implements only loudness, and
// the sink advertises only loudness).  Refuses -- returns 0 and counts -- a bad sync, a CRC mismatch, a truncated
// frame or an unsupported header (SNR allocation, 4 subbands), so a corrupt frame never lands in the ring.
// MIT.  Nothing here was derived from any existing implementation.
#pragma once
#include <stdint.h>
#include "Sbc.h"
struct SbcDecoder {
    // Header only: sync + byte 1 + bitpool -> Params.  false = not an SBC frame header.
    static bool parseHeader(const uint8_t *f, uint16_t len, Sbc::Params &p);
    // Decode ONE frame at f (len bytes available).  Returns the frame length consumed (0 = refused).  left/right get
    // blocks*8 samples each (128 at 16 blocks); MONO copies left into right.
    uint16_t decode(const uint8_t *f, uint16_t len, int16_t *left, int16_t *right);
    void reset();                                    // clear the synthesis state (a new stream)
    uint32_t badSync()     const { return m_badSync; }
    uint32_t crcErrors()   const { return m_crc; }
    uint32_t shortFrames() const { return m_short; }
    uint32_t unsupported() const { return m_unsup; }
private:
    float m_v[2][160] = {{0}};                       // synthesis state per channel (V, newest first)
    uint32_t m_badSync = 0, m_crc = 0, m_short = 0, m_unsup = 0;
    void synthesise(uint8_t ch, const int32_t *sub /*8 subband samples, Q15-scaled like the encoder's*/, int16_t *out8);
};
```

```cpp
// Synthesis output scale.  The encoder scales its analysis output by 16384 (= 32768/2) so a unity round trip through
// the spec's synthesis needs the inverse: subband samples are Q15-ish (|x| <= 32768 at full scale), the synthesis
// sums 10 windowed taps of 16 cosine-mixed values.  8.0f * (1/16384) * 32768/32768... -- pinned EMPIRICALLY by
// sbcdecoder_test scenario 3 (unity within 5 %): start at 8.0f / 16384.0f * 32768.0f = 16.0f; the two admissible
// alternatives if the round trip lands a power of two off are 8.0f and 32.0f.  Record the measured one here.
static const float SYNTH_SCALE = 16.0f;
bool SbcDecoder::parseHeader(const uint8_t *f, uint16_t len, Sbc::Params &p) {
    if (len < 4 || f[0] != 0x9C) return false;
    p.rate = (Sbc::Rate)(f[1] >> 6); p.blocks = (uint8_t)(((f[1] >> 4) & 3) * 4 + 4); p.mode = (Sbc::Mode)((f[1] >> 2) & 3);
    p.alloc = (Sbc::Alloc)((f[1] >> 1) & 1); p.subbands = (f[1] & 1) ? 8 : 4; p.bitpool = f[2];
    return p.bitpool >= 2;
}
void SbcDecoder::reset() { memset(m_v, 0, sizeof m_v); }
// One block of one channel: 8 subband samples in -> 8 PCM samples out (spec 12.6.4, 8-subband synthesis).
void SbcDecoder::synthesise(uint8_t ch, const int32_t *sub, int16_t *out8) {
    static float N[16][8]; static bool init = false;
    if (!init) { for (int i = 0; i < 16; i++) for (int k = 0; k < 8; k++) N[i][k] = cosf((i + 4) * (2 * k + 1) * (float)M_PI / 16.0f); init = true; }
    float *V = m_v[ch];
    for (int i = 159; i >= 16; i--) V[i] = V[i - 16];                                   // shift by 16
    for (int i = 0; i < 16; i++) { float s = 0; for (int k = 0; k < 8; k++) s += N[i][k] * (float)sub[k] / 32768.0f; V[i] = s; }
    float U[80];
    for (int i = 0; i < 5; i++) for (int j = 0; j < 8; j++) { U[i * 16 + j] = V[i * 32 + j]; U[i * 16 + j + 8] = V[i * 32 + j + 24]; }
    float W[80]; for (int i = 0; i < 80; i++) W[i] = U[i] * PROTO8[i] * SYNTH_SCALE;
    for (int j = 0; j < 8; j++) { float s = 0; for (int i = 0; i < 10; i++) s += W[j + 8 * i];
        int32_t v = (int32_t)lrintf(s * 32768.0f); if (v > 32767) v = 32767; if (v < -32768) v = -32768; out8[j] = (int16_t)v; }
}
uint16_t SbcDecoder::decode(const uint8_t *f, uint16_t len, int16_t *L, int16_t *R) {
    Sbc::Params p;
    if (len < 4 || f[0] != 0x9C) { m_badSync++; return 0; }
    if (!parseHeader(f, len, p)) { m_badSync++; return 0; }
    if (p.subbands != 8 || p.alloc != Sbc::LOUDNESS || p.bitpool > 250) { m_unsup++; return 0; }
    uint16_t flen = Sbc::frameLength(p);
    if (flen > len) { m_short++; return 0; }
    int ch = (p.mode == Sbc::MONO) ? 1 : 2; int blocks = p.blocks;
    // --- bit reader over the frame body ---
    uint32_t pos = 32;                                                       // bit position after the 4 header bytes
    auto get = [&](int nbits) -> uint32_t { uint32_t v = 0; for (int i = 0; i < nbits; i++) { v = (v << 1) | ((f[pos >> 3] >> (7 - (pos & 7))) & 1); pos++; } return v; };
    uint8_t join = 0; if (p.mode == Sbc::JOINT_STEREO) join = (uint8_t)get(8);
    uint8_t sf[2][8] = {{0}}, sfN[16]; int k = 0;
    for (int c = 0; c < ch; c++) for (int s = 0; s < 8; s++) { sf[c][s] = (uint8_t)get(4); sfN[k++] = sf[c][s]; }
    if (f[3] != Sbc::crc8(f + 1, 2, sfN, (uint8_t)(ch * 8), join, p)) { m_crc++; return 0; }
    uint8_t bits[2][8] = {{0}}; Sbc::allocateBits(p, sf, bits);
    // --- dequantise: x = 2^(sf+1) * ((2q+1)/levels - 1)  (spec 12.6.2 inverse), integer-exact ---
    static int32_t sub[2][16][8];
    for (int b = 0; b < blocks; b++) for (int c = 0; c < ch; c++) for (int s = 0; s < 8; s++) {
        if (!bits[c][s]) { sub[c][b][s] = 0; continue; }
        int32_t levels = (1 << bits[c][s]) - 1; int64_t q = get(bits[c][s]);
        int64_t x = ((((2 * q + 1) << (sf[c][s] + 1)) / levels) - (1LL << (sf[c][s] + 1)));
        sub[c][b][s] = (int32_t)x;
    }
    if (p.mode == Sbc::JOINT_STEREO)                                         // M/S -> L/R for the joined subbands (encoder: m=(l+r)/2, d=(l-r)/2)
        for (int s = 0; s < 8; s++) if (join & (0x80 >> s)) for (int b = 0; b < blocks; b++) {
            int32_t m = sub[0][b][s], d = sub[1][b][s]; sub[0][b][s] = m + d; sub[1][b][s] = m - d; }
    for (int b = 0; b < blocks; b++) { synthesise(0, sub[0][b], L + b * 8); if (ch == 2) synthesise(1, sub[1][b], R + b * 8); }
    if (ch == 1) memcpy(R, L, (size_t)blocks * 8 * sizeof(int16_t));
    return flen;
}
```

```cpp
void A2dpSink::tick(uint32_t now) {
    m_link.tick(now);
    if (m_st != IDLE && m_st != DONE && m_st != DISCONNECTING && m_link.lost()) {
        m_link.ackLost(); m_avdtp.reset(); m_avrcp.reset(); m_l2.reset();
        logf("sink: link lost reason=0x%02X", m_link.lostReason()); m_result = LOST; m_st = DONE; return;
    }
    switch (m_st) {
    case PAIRING:
        if (m_link.busy()) return;
        if (!m_opIssued) { m_opIssued = m_link.startPair(true); return; }
        if (m_link.result() != BtLink::OK) { m_result = PAIR_FAILED; m_st = DISCONNECTING; break; }
        m_l2.begin(m_link.handle(), m_aclNum); m_l2.acceptIncoming(true);
        m_l2.allowPsm(Avdtp::PSM); m_l2.allowPsm(Sdp::PSM); m_l2.allowPsm(Avrcp::PSM); m_l2.onData(onData, this);
        m_avdtp.begin(m_l2, 0, 0); m_avdtp.setLocalSep(Avdtp::SEP_SINK);
        m_st = AVDTP_WAIT; m_deadline = now + 20000; break;                                 // a phone opens AVDTP within seconds; iOS sometimes after AVRCP
    case AVDTP_WAIT:
        m_avdtp.adoptInbound(m_l2);
        if (m_avdtp.role() == Avdtp::ACCEPTOR) { m_st = AVDTP; m_deadline = now + 30000; break; }
        if ((int32_t)(now - m_deadline) >= 0) { m_result = L2CAP_FAILED; m_st = DISCONNECTING; }
        break;
    case AVDTP:
        m_avdtp.adoptInbound(m_l2);
        if (m_avdtp.configChanged()) adoptConfig();
        if (m_avdtp.started()) { m_result = OK; m_st = STREAMING; logf("sink: streaming bitpool=%u", m_params.bitpool); break; }
        if (m_avdtp.state() == Avdtp::FAILED || (int32_t)(now - m_deadline) >= 0) { m_result = AVDTP_FAILED; m_st = DISCONNECTING; }
        break;
    case STREAMING:
        m_avdtp.adoptInbound(m_l2);
        if (!m_delaySent && m_avdtp.peerWantsDelayReports() && m_avdtp.sendDelayReport(m_delayTenthMs)) { m_delaySent = true; logf("sink: delay_report=%u", m_delayTenthMs); }
        break;
    case DISCONNECTING:

    default: break;
    }
    m_l2.service(); m_avdtp.service(); m_sdpServer.service(m_l2); m_avrcp.service(m_l2);
}
```

```cpp
// inquiry scan until a bond exists or always if asked), take over each accepted incoming link as an A2dpSink attempt,

private:
    A2dpSink &m_sink; BondTable *m_bonds = nullptr; State m_state = IDLE; bool m_alwaysDisc = false; Stats m_stats{};
```

```c
// The A2DP sink's clock servo (NEW-41), pure C so the host test can drive it.  Input: the PCM ring's fill in blocks,
// sampled once per audio block (every 128 samples).  Output: the PLL trim in ppm.  Proportional on FILL -- which is
// already the integral of the rate error -- with a first-order filter on the fill (tau ~1 s = 344 blocks) so packet
// bursts do not move the pitch, a +-200 ppm clamp, and hold (no update) when told the link is down.
#pragma once
#include <stdint.h>
typedef struct { int32_t target_x256; int32_t filt_x256; int32_t trim_ppm; int32_t kp_ppm_per_block_x256; int32_t clamp_ppm; } sink_servo_t;
static inline void servo_init(sink_servo_t *s, int32_t target_blocks) {
    s->target_x256 = target_blocks * 256; s->filt_x256 = s->target_x256; s->trim_ppm = 0;
    s->kp_ppm_per_block_x256 = 40 * 256;    /* 40 ppm per block of fill error: +-200 ppm at +-5 blocks, inside a 16-block ring */
    s->clamp_ppm = 200;
}
/* one audio block elapsed; fill = blocks currently in the ring (0..ring size); hold = link down (freeze the trim) */
static inline int32_t servo_step(sink_servo_t *s, int32_t fill, int hold) {
    if (hold) return s->trim_ppm;
    s->filt_x256 += ((fill * 256) - s->filt_x256) / 344;                          /* EMA, tau = 344 blocks = 1 s */
    int32_t err_x256 = s->filt_x256 - s->target_x256;                              /* + = ring filling: source faster than us -> speed up */
    int32_t t = (int32_t)(((int64_t)err_x256 * s->kp_ppm_per_block_x256) / (256 * 256));
    if (t > s->clamp_ppm) t = s->clamp_ppm; if (t < -s->clamp_ppm) t = -s->clamp_ppm;
    s->trim_ppm = t; return t;
}
static inline void servo_recentre(sink_servo_t *s) { s->filt_x256 = s->target_x256; }   /* a new START: drop the history, keep the trim */
```

```cpp
static uint32_t crc32Update(uint32_t c, const uint8_t *p, size_t n) { for (size_t i = 0; i < n; i++) { c ^= p[i]; for (int k = 0; k < 8; k++) c = (c >> 1) ^ (0xEDB88320u & (0u - (c & 1u))); } return c; }
void AudioInputBluetooth::begin() {
    m_head = m_tail = 0; m_dec.reset(); m_haveSeq = false; m_fragLen = 0;
    servo_init(&m_servo, TARGET); m_live = true; audioPllTrimPpm(m_servo.trim_ppm);
}
void AudioInputBluetooth::end() { m_live = false; m_head = m_tail = 0; m_fragLen = 0; }   // the trim HOLDS (servo_step is skipped while !m_live)
void AudioInputBluetooth::pushFrame(const uint8_t *f, uint16_t len) {
    uint16_t head = m_head, next = (uint16_t)((head + 1) % RING);
    if (next == m_tail) { m_over++; m_tail = (uint16_t)((m_tail + 1) % RING); }          // overflow: drop the OLDEST block (keep latency bounded)
    if (m_dec.decode(f, len, m_ring[head].l, m_ring[head].r) == 0) { m_bad++; return; }
    if (m_crcBlocks < 200) { m_crc = crc32Update(m_crc, (const uint8_t *)m_ring[head].l, sizeof m_ring[head].l); m_crcBlocks++; }
    for (int i = 0; i < AUDIO_BLOCK_SAMPLES; i++) { int32_t v = m_ring[head].l[i]; m_rmsAcc += (uint32_t)(v < 0 ? -v : v); } m_rmsBlocks++;
    m_head = next; m_frames++;
}
void AudioInputBluetooth::onMedia(const uint8_t *p, uint16_t len) {
    if (!m_live || len < 13) return;
    if ((p[0] >> 6) != 2) { m_bad++; return; }                                             // RTP v2
    uint16_t hdr = (uint16_t)(12 + 4 * (p[0] & 0x0F));                                     // CSRC count
    if (len < hdr + 1) { m_bad++; return; }
    uint16_t seq = (uint16_t)((p[2] << 8) | p[3]);
    if (m_haveSeq && seq != (uint16_t)(m_lastSeq + 1)) m_seqGaps++;
    m_lastSeq = seq; m_haveSeq = true; m_pkts++;
    uint8_t mh = p[hdr]; bool F = mh & 0x80, S = mh & 0x40, L = mh & 0x20; uint8_t n = (uint8_t)(mh & 0x0F);
    const uint8_t *q = p + hdr + 1; uint16_t rem = (uint16_t)(len - hdr - 1);
    if (F) {                                                                               // a fragmented frame: gather until L
        if (S) m_fragLen = 0;
        if (m_fragLen + rem <= sizeof m_frag) { memcpy(m_frag + m_fragLen, q, rem); m_fragLen = (uint16_t)(m_fragLen + rem); } else { m_fragLen = 0; m_bad++; }
        if (L && m_fragLen) { pushFrame(m_frag, m_fragLen); m_fragLen = 0; }
        return;
    }
    for (uint8_t i = 0; i < n && rem >= 4; i++) {                                          // n whole frames back to back
        Sbc::Params fp; if (!SbcDecoder::parseHeader(q, rem, fp)) { m_bad++; break; }
        uint16_t fl = Sbc::frameLength(fp); if (fl > rem) { m_bad++; break; }
        pushFrame(q, fl); q += fl; rem = (uint16_t)(rem - fl);
    }
}
void AudioInputBluetooth::update(void) {
    audio_block_t *l = allocate(), *r = allocate();
    if (!l || !r) { if (l) release(l); if (r) release(r); return; }
    uint16_t tail = m_tail;
    if (!m_live || tail == m_head) { memset(l->data, 0, sizeof l->data); memset(r->data, 0, sizeof r->data); if (m_live) m_under++; }
    else { memcpy(l->data, m_ring[tail].l, sizeof l->data); memcpy(r->data, m_ring[tail].r, sizeof r->data); m_tail = (uint16_t)((tail + 1) % RING); }
    if (m_live) { int32_t t = servo_step(&m_servo, fill(), 0); static int32_t applied = 0; if (t != applied) { applied = t; audioPllTrimPpm(t); } }
    transmit(l, 0); transmit(r, 1); release(l); release(r);
}
```
